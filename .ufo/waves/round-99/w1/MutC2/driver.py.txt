def sh(root, *args):
    return subprocess.run(["git", "-C", str(root), *args], capture_output=True, text=True, env=ENV)


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest() if p.exists() else "absent"


def lines(p):
    return p.read_text().splitlines(keepends=True)


def write_lines(p, ls):
    p.write_text("".join(ls))


def find_line(ls, prefix):
    hits = [i for i, l in enumerate(ls) if l.startswith(prefix)]
    assert len(hits) == 1, (prefix, hits)
    return hits[0]


def rep(p, old, new, count=1):
    t = p.read_text()
    assert t.count(old) == count, (str(p), old[:40], t.count(old))
    p.write_text(t.replace(old, new))

def run_counter(rev, work):
    if work.exists():
        shutil.rmtree(work)
    proc = subprocess.run([sys.executable, "plans/round99_c2_rowscoped_bind.py", "--lane", ".", "--rev", rev,
                           "--work", str(work)], cwd=str(FX), capture_output=True, text=True, env=ENV, timeout=600)
    return proc.returncode, proc.stdout + (("\n[stderr]\n" + proc.stderr) if proc.stderr.strip() else "")


def check_pristine():
    top = sh(FX, "rev-parse", "--show-toplevel").stdout.strip()
    assert top == str(FX), (top, str(FX))
    head = sh(FX, "rev-parse", "HEAD").stdout.strip()
    assert head == T, head
    st = sh(FX, "status", "--porcelain").stdout.strip()
    assert st == "", st


def main(ids):
    check_pristine()
    summary = []
    for mid in ids:
        desc, fn, files = MUT[mid]
        check_pristine()
        before = {f: sha(FX / f) for f in files}
        fn()
        after = {f: sha(FX / f) for f in files}
        assert before != after, mid
        add = sh(FX, "add", "-A")
        assert add.returncode == 0, add.stderr
        c = sh(FX, "-c", "user.name=mutant", "-c", "user.email=mutant@local", "commit", "-q", "-m", f"mutant {mid}: {desc}")
        assert c.returncode == 0, c.stderr
        rev = sh(FX, "rev-parse", "HEAD").stdout.strip()
        rc, out = run_counter(rev, ROOT / f"w-{mid}")
        (OUT / f"{mid}.txt").write_text(f"# {mid}: {desc}\n# rev {rev}\n# rc {rc}\n# before {json.dumps(before)}\n# after {json.dumps(after)}\n\n{out}")
        legs = next((l for l in out.splitlines() if l.startswith("LEGS ")), "LEGS -")
        pred = next((l for l in out.splitlines() if l.startswith("PREDICATE ")), "PREDICATE -")
        defects = [l for l in out.splitlines() if l.startswith("DEFECT ")]
        summary.append({"id": mid, "desc": desc, "rev": rev, "rc": rc, "legs": legs, "predicate": pred,
                        "defects": defects, "before": before, "after": after})
        print(mid, rc, legs, pred, "|", " ; ".join(defects)[:300], flush=True)
        sh(FX, "reset", "--hard", "-q", T)
        sh(FX, "clean", "-fdq")
        check_pristine()
    (OUT / "summary.json").write_text(json.dumps(summary, indent=1))
    print("SUMMARY", OUT / "summary.json")


if __name__ == "__main__":
    main(sys.argv[1:])
